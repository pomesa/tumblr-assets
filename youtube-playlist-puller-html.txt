(() => {
    return [...document.querySelectorAll("ytd-playlist-video-renderer")]
        .map(video => {
            const link = video.querySelector("a#video-title");
            if (!link) return null;

            return {
                title: link.textContent.trim(),
                url: link.href,
                id: new URL(link.href).searchParams.get("v")
            };
        })
        .filter(Boolean);
})();
