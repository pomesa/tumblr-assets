
(() => {
    const videos = [...document.querySelectorAll("yt-lockup-view-model")]
        .map(video => {
            const link = video.querySelector(
                "a.ytLockupMetadataViewModelTitle, h3 a"
            );

            if (!link) return null;

            const url = new URL(link.href, location.origin);

            return {
                id: url.searchParams.get("v"),
                title: (link.title || link.textContent).trim()
            };
        })
        .filter(v => v && v.id);

    console.log(
        videos
            .map(v => `{ id: "${v.id}", title: "${v.title.replace(/\\/g, '\\\\').replace(/"/g, '\\"')}" },`)
            .join("\n")
    );

    return videos;
})();
