(() => {
    const videos = [...document.querySelectorAll("ytd-playlist-video-renderer")]
        .map(video => {
            const link = video.querySelector("a#video-title");
            if (!link) return null;

            return {
                id: new URL(link.href).searchParams.get("v"),
                title: (link.title || link.textContent).trim()
            };
        })
        .filter(v => v && v.id);

    console.log(
        videos
            .map(v => `{ id: "${v.id}", title: "${v.title.replace(/"/g, '\\"')}" },`)
            .join("\n")
    );

    return videos;
})();
